51template <typename T>
52void ArrayList<T>::addSpace() {
53 /* Reallocate the capacity of the buffer (similar to realloc in C)
54 * Typically that of double the original capacity
55 */
56 int newCapacity = (capacity - data) * 2;
57
58 // If the original capacity is 0, set it to 1 to be minimal and memory efficient
59 if(newCapacity == 0) {
60 newCapacity = 1;
61 }
62
63 // Allocate a new data buffer that is set to the updated capacity
64 T* newData = new T[newCapacity];
65
66 // Calculate the number of elements currently in the `ArrayList`
67 int currentSize = size - data;
68
69 // Copy all contents from the old data buffer into the new data buffer
70 for(int i = 0; i < currentSize; i++) {
71 newData[i] = data[i];
72 }
73
74 // Free the contents from the old buffer from memory
75 delete[] data;
76
77 // Update the data pointer to point to the newly allocated buffer
78 data = newData;
79
80 // Point `capacity` to the end of the new buffer
81 capacity = data + newCapacity;
82
83 // Update the size pointer to point to the correct position in the new buffer
84 size = data + currentSize;
85}
