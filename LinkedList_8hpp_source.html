33ListNode<T>*& LinkedList<T>::find(int index) {
34 // return the head if the index is 0
35 // we know that the first node in the list is always the head!
36 if(index == 0) {
37 return head;
38 }
39
40 // create a copy pointer of the head to traverse through the list!
41 ListNode<T>* curr = head;
42
43 // traverse through the list to find the node right before the given index
44 int x = 0;
45 while(curr != nullptr && x < index - 1) {
46 curr = curr->next;
47 x += 1;
48 }
49
50 // return the next node in the list, since that matches the node at the given index
51 return curr->next;
