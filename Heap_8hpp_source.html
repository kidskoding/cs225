74void Heap<T>::_insert(const T& key) {
75 // check to ensure there's space to insert an element
76 // ...if not, grow the array
77 if(size_ == capacity_) { _growArray(); }

89template <class T>
90void Heap<T>::_heapifyUp(size_t index) {
91 // index 1 is the root, which has no parent to swap with
92 if(index > 1) {
93 // swap with the parent if it is smaller, then keep going up
94 if(item_[index] < item_[_parent(index)]) {
95 std::swap(item_[index], item_[_parent(index)]);
96 _heapifyUp(_parent(index));
97 }
98 }
