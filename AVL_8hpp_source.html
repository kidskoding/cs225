21template <typename K, typename V>
22void AVL<K, V>::_ensureBalance(TreeNode*& curr) {
23 // calculate the balance factor
24 int balance = height(curr->right) - height(curr->left);
25
26 // check if the node is currently not in balance
27 if(balance == -2) {
28 int leftBalance = height(curr->left->right) - height(curr->left->left);
29 if(leftBalance == -1) { // heavy left ("left stick" shaped subtree)
30 _rotateRight(curr);
31 } else { // "left elbow" shaped subtree
32 _rotateLeftRight(curr);
33 }
34 } else if(balance == 2) {
35 int rightBalance = height(curr->right->right) - height(curr->right->left);
36 if(rightBalance == -1) { // heavy right ("right stick" shaped subtree)
37 _rotateLeft(curr);
38 } else { // "right elbow" shaped subtree
39 _rotateRightLeft(curr);
40 }
41 }
42
43 // update the height of each node after performing rotations
44 _updateHeight(curr);
45}

51void AVL<K, V>::_updateHeight(TreeNode* curr) {
52 // calculate height as 1 + maximum height of children
53 // the +1 accounts for the extra edge needed to descend into the subtree with a greater height
54 // this works after any tree modification because height depends on the current tree structure
55 if(curr != nullptr) {
56 curr->height = 1 + std::max(height(curr->left), height(curr->right));
