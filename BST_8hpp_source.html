10template <typename K, typename V>
11typename BST<K, V>::TreeNode*& BST<K, V>::_find(TreeNode*& root, const K& key) {
12 // if the root's pointer is null, return the root or
13 // if the root's key equals the given key, return the root
14 if(root == nullptr || root->key == key) { return root; }
15
16 // if the root's key is greater than the given key, call find on the left subtree
17 if(root->key > key) {
18 return _find(root->left, key);
19 }
20
21 // if the root's key is less than the given key, call find on the right subtree
22 return _find(root->right, key);
23}

37template <class K, class V>
38void BST<K, V>::_insert(TreeNode*& root, const K& key, const V& value) {
39 // grab the reference/location to the pointer where `key` should be inserted
40 TreeNode*& loc = _find(root, key);
41
42 // the notes assume keys are unique; if `key` is already here, just update its value
43 // (overwriting `loc` would throw away the node and both of its subtrees)
44 if(loc != nullptr) {
45 loc->value = value;
46 return;
47 }
48
49 // assign `loc` to point to a new `TreeNode` with the given `key` and `value`
50 loc = new TreeNode(key, value);
51}

67template <typename K, typename V>
68typename BST<K, V>::TreeNode*& BST<K, V>::_iop(TreeNode*& root) {
69 // grab a pointer to the left subtree pointer of the root
70 // (a pointer to the pointer, so moving `curr` never overwrites a link in the tree)
71 TreeNode** curr = &root->left;
72
73 // iterate through each right child of the left subtree to find the rightmost node
74 // return the rightmost node (since that always contains the greatest value)
75 while((*curr)->right != nullptr) {
76 curr = &(*curr)->right;
77 }
78 return *curr;
79}

94template <typename K, typename V>
95void BST<K, V>::_remove(TreeNode*& node, const K& key) {
96 if(node == nullptr) { return; }
97
98 if(key < node->key) {
99 // descend down the left subtree because left subtree contains the smaller values
100 _remove(node->left, key);
101 } else if(key > node->key) {
102 // descend down the right subtree because right subtree contains the larger values
103 _remove(node->right, key);
104 } else {
105 // case 1: the leaf node case - free the node and set it to null
106 if(node->left == nullptr && node->right == nullptr) {
107 delete node;
108 node = nullptr;
109 }
110
111 // case 2: the one child node case (either left or right child)
112 else if(node->left == nullptr) {
113 // remove the node by connecting its parent directly to the node's only child (the right child)
114 TreeNode* tmp = node;
115 node = node->right;
116 delete tmp;
117 }
118 else if(node->right == nullptr) {
119 // remove the node by connecting its parent directly to the node's only child (the left child)
120 TreeNode* tmp = node;
121 node = node->left;
122 delete tmp;
123 }
124
125 // case 3: the two child node case (both left and right child)
126 // swap the removed node with the iop and recursively call the function on the left subtree
127 else {
128 K iop_key = node->key;
129 TreeNode* iop = _iop(node);
130 _swap(iop, node);
131
132 // recursively call on the left subtree because the iop (the node to remove) is located there
133 _remove(node->left, iop_key);
134 }
135 }
136}
